                            <h4 class="text-lg font-bold text-stone-800 mb-3 underline decoration-blue-200 underline-offset-4">1. Total Inductance ($L_{total}$) 구하기</h4>
                            <p class="text-stone-600 leading-relaxed mb-4">
                                실제 커패시터의 총 인덕턴스는 부품 자체의 기생 인덕턴스(ESL)와 PCB 실장 시 발생하는 루프 인덕턴스의 합입니다.
                            </p>
                            <div class="bg-stone-50 p-6 rounded-xl border border-stone-100 shadow-inner flex justify-center items-center">
                                $$L_{total} = ESL + L_{loop}$$
                            </div>
                            <ul class="mt-4 space-y-2 text-stone-600 text-sm">
                                <li class="flex items-start"><span class="font-bold text-blue-600 mr-2">ESL:</span> 데이터시트에 명시된 부품 내부 값.</li>
                                <li class="flex items-start"><span class="font-bold text-amber-600 mr-2">$L_{loop}$:</span> 전류가 흐르는 경로(Trace, Via)와 그 아래 Ground Plane 사이의 루프 면적에 비례합니다.</li>
                            </ul>
                        </section>

                        <section>
                            <h4 class="text-lg font-bold text-stone-800 mb-3 underline decoration-blue-200 underline-offset-4">2. Total Impedance ($Z_{total}$) 구하기</h4>
                            <p class="text-stone-600 leading-relaxed mb-4">
                                커패시터, 저항(ESR), 인덕터(Total L)가 직렬로 연결된 RLC 회로로 모델링됩니다.
                            </p>
                            <div class="bg-stone-50 p-6 rounded-xl border border-stone-100 shadow-inner flex justify-center items-center">
                                $$|Z| = \sqrt{ESR^2 + (2\pi f L_{total} - \frac{1}{2\pi f C})^2}$$
                            </div>
                            <p class="mt-4 text-stone-600 leading-relaxed">
                                주파수가 공진점($f_{res}$)을 넘어서면, 임피던스는 $2\pi f L_{total}$에 의해 지배되어 주파수에 비례해 증가합니다(Inductive region). 
                                즉, $L_{loop}$가 클수록 고주파 노이즈 제거 성능이 떨어집니다.
                            </p>

    <script>
        // State Management
        const state = {
            cap: 100e-9,    // 기본 100nF
            esr: 0.05,      // 50mOhm
            esl: 0.5e-9,    // 0.5nH
            traceLen: 5.0,  // mm
            h: 0.5,         // mm
            loopL: 0,
            totalL: 0,
            srf: 0
        };

        const MU_0 = 4 * Math.PI * 1e-7;
        let impedanceChart = null;

        // Core Calculations
        function calculatePhysics() {
            const l_m = state.traceLen / 1000;
            const h_m = state.h / 1000;
            const w_m = 0.5 / 1000; // 배선 폭은 0.5mm로 고정하여 계산
            
            // 루프 인덕턴스 근사식
            state.loopL = ((MU_0 * h_m * l_m) / w_m) + 0.4e-9;
            state.totalL = state.esl + state.loopL;
            
            // 자기 공진 주파수 (SRF)
            state.srf = 1 / (2 * Math.PI * Math.sqrt(state.totalL * state.cap));
        }

        function updateLabels() {
            // 용량 표시 로직: uF와 nF 구분
            let displayCap = "";
            if (state.cap >= 1e-6) {
                displayCap = (state.cap * 1e6).toFixed(1) + ' µF';
            } else {
                displayCap = (state.cap * 1e9).toFixed(0) + ' nF';
            }
            document.getElementById('val-cap').textContent = displayCap;
            
            document.getElementById('val-esr').textContent = (state.esr * 1000).toFixed(0) + ' mΩ';
            document.getElementById('val-esl').textContent = (state.esl * 1e9).toFixed(1) + ' nH';
            document.getElementById('val-len').textContent = state.traceLen.toFixed(1) + ' mm';
            document.getElementById('val-height').textContent = state.h.toFixed(1) + ' mm';
            
            document.getElementById('res-loop-l').textContent = (state.loopL * 1e9).toFixed(2);
            document.getElementById('res-total-l').textContent = (state.totalL * 1e9).toFixed(2);

            // Update visuals
            const hPx = state.h * 20 + 10;
            document.getElementById('vis-height').style.height = `${hPx}px`;
            document.getElementById('vis-via1').style.height = `${hPx}px`;
            document.getElementById('vis-via2').style.height = `${hPx}px`;
            document.getElementById('vis-trace').style.bottom = `${8 + hPx}px`;

            // 루프 길이에 따라 시각화 너비 조정
            // traceLen (1-50mm)을 시각적 너비 (10%-80%)에 매핑합니다.
            const traceWidthPercent = ((state.traceLen - 1) / (50 - 1)) * (80 - 10) + 10;
            // 트레이스는 중앙에 위치하므로 left와 right는 (100 - 너비) / 2입니다.
            const traceInsetPercent = (100 - traceWidthPercent) / 2;
            
            document.getElementById('vis-trace').style.left = `${traceInsetPercent}%`;
            document.getElementById('vis-trace').style.right = `${traceInsetPercent}%`;
            document.getElementById('vis-via1').style.left = `${traceInsetPercent}%`;
            document.getElementById('vis-via2').style.right = `${traceInsetPercent}%`;
            
            // SRF 표시 로직: Hz, kHz, MHz 구분
            let srfVal = state.srf;
            if (srfVal >= 1e6) {
                document.getElementById('res-srf').textContent = (srfVal / 1e6).toFixed(2);
            } else {
                // MHz 단위로 통일하되, 값이 작을 경우 소수점 표시
                document.getElementById('res-srf').textContent = (srfVal / 1e6).toFixed(4);
            }
        }

        function updateChart() {
            const dataPoints = [];
            const idealPoints = [];
            
            // 용량이 커지면 공진점이 매우 낮아지므로 범위를 넓게 조정 (10Hz ~ 1GHz)
            let startHz = 100;
            if (state.cap > 10e-6) startHz = 10; // 10uF 이상은 10Hz부터 시작
            
            for (let exp = Math.log10(startHz); exp <= 9.2; exp += 0.05) {
                const f = Math.pow(10, exp);
                const w = 2 * Math.PI * f;
                
                // 실제 임피던스
                const z = Math.sqrt(Math.pow(state.esr, 2) + Math.pow((w * state.totalL - 1/(w * state.cap)), 2));
                dataPoints.push({x: f, y: z});

                // 인덕턴스가 없는 이상적인 경우
                const z_id = Math.sqrt(Math.pow(state.esr, 2) + Math.pow((-1/(w * state.cap)), 2));
                idealPoints.push({x: f, y: z_id});
            }

            impedanceChart.data.datasets[0].data = dataPoints;
            impedanceChart.data.datasets[1].data = idealPoints;
            
            // X축 최소값 동적 변경
            impedanceChart.options.scales.x.min = startHz;
            
            impedanceChart.update('none');
        }

        function handleInput(e) {
            const id = e.target.id;
            const val = parseFloat(e.target.value);
            
            if (id === 'input-cap') state.cap = val * 1e-9;
            if (id === 'input-esr') state.esr = val * 1e-3;
            if (id === 'input-esl') state.esl = val * 1e-9;
            if (id === 'input-len') state.traceLen = val;
            if (id === 'input-height') state.h = val;

            calculatePhysics();
            updateLabels();
            updateChart();
        }

        function init() {
            const inputs = document.querySelectorAll('input[type="range"]');
            inputs.forEach(input => input.addEventListener('input', handleInput));

            const ctx = document.getElementById('impedanceChart').getContext('2d');
            impedanceChart = new Chart(ctx, {
                type: 'line',
                data: {
                    datasets: [
                        {
                            label: '실제 $|Z|$ (Total L 반영)',
                            borderColor: '#3b82f6',
                            borderWidth: 3,
                            data: [],
                            pointRadius: 0,
                            tension: 0.1
                        },
                        {
                            label: '이상적 $|Z|$ ($L=0$)',
                            borderColor: '#e2e8f0',
                            borderDash: [5, 5],
                            borderWidth: 2,
                            data: [],
                            pointRadius: 0,
                            tension: 0.1
                        }
                    ]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    scales: {
                        x: {
                            type: 'logarithmic',
                            title: { display: true, text: '주파수 (Hz)' },
                            min: 100,
                            max: 1500000000
                        },
                        y: {
                            type: 'logarithmic',
                            title: { display: true, text: '임피던스 (Ω)' },
                            min: 0.001,
                            max: 1000
                        }
                    },
                    plugins: {
                        legend: { position: 'bottom' }
                    }
                }
            });

            calculatePhysics();
            updateLabels();
            updateChart();
        }

        window.onload = init;
    </script>
